# generate function to determine the Subsystems
def assign_subsystem(df):
    # marine
    wetlands.loc[lambda df: df["C Code"].str.startswith("M1", na = False), "C Subsystem"] = 'Subtidal'
    wetlands.loc[lambda df: df["C Code"].str.startswith("M2", na = False), "C Subsystem"] = 'Intertidal' 
    # estuarine
    wetlands.loc[lambda df: df["C Code"].str.startswith("E1", na = False), "C Subsystem"] = 'Subtidal'
    wetlands.loc[lambda df: df["C Code"].str.startswith("E2", na = False), "C Subsystem"] = 'Intertidal' 
    # riverine
    wetlands.loc[lambda df: df["C Code"].str.startswith("R1", na = False), "C Subsystem"] = 'Tidal'
    wetlands.loc[lambda df: df["C Code"].str.startswith("R2", na = False), "C Subsystem"] = 'Lower Perennial'
    wetlands.loc[lambda df: df["C Code"].str.startswith("R3", na = False), "C Subsystem"] = 'Upper Perennial'
    wetlands.loc[lambda df: df["C Code"].str.startswith("R4", na = False), "C Subsystem"] = 'Intermittent'
    wetlands.loc[lambda df: df["C Code"].str.startswith("R5", na = False), "C Subsystem"] = 'Unknown Perennial'

    # lacustrine
    wetlands.loc[lambda df: df["C Code"].str.startswith("L1", na = False), "C Subsystem"] = 'Limnetic'
    wetlands.loc[lambda df: df["C Code"].str.startswith("L2", na = False), "C Subsystem"] = 'Littoral'
    # palustrine
    wetlands.loc[lambda df: df["C Code"].str.startswith("P", na = False), "C Subsystem"] = "N/A"
    
    return df

## apply the subsystem function

# this code was generated with the help of chatgpt to best understand the manipulation of a sankey diagram.

df = wetlands.copy()

df["Hectares"] = pd.to_numeric(df["Hectares"], errors="coerce")
df = df.dropna(subset=["Hectares"])

cols = ["C System", "C Subsystem", "C Class"]

for col in cols:
    df[col] = df[col].fillna("Unknown").astype(str)

# 2. Calculate total hectares for each transition
flow1 = (
    df.groupby(["C System", "C Subsystem"])["Hectares"]
    .sum()
    .reset_index()
)

flow2 = (
    df.groupby(["C Subsystem", "C Class"])["Hectares"]
    .sum()
    .reset_index()
)

# 3. Create separate nodes for each classification level
systems = df["C System"].unique().tolist()
subsystems = df["C Subsystem"].unique().tolist()
classes = df["C Class"].unique().tolist()

labels = (
    [f"System: {x}" for x in systems]
    + [f"Subsystem: {x}" for x in subsystems]
    + [f"Class: {x}" for x in classes]
)

system_map = {x: i for i, x in enumerate(systems)}

subsystem_map = {
    x: i + len(systems)
    for i, x in enumerate(subsystems)
}

class_map = {
    x: i + len(systems) + len(subsystems)
    for i, x in enumerate(classes)
}

# 4. Build links
sources = (
    flow1["C System"].map(system_map).tolist()
    + flow2["C Subsystem"].map(subsystem_map).tolist()
)

targets = (
    flow1["C Subsystem"].map(subsystem_map).tolist()
    + flow2["C Class"].map(class_map).tolist()
)

values = (
    flow1["Hectares"].tolist()
    + flow2["Hectares"].tolist()
)

# 5. Generate the Sankey diagram
fig_subset = go.Figure(go.Sankey(
    node=dict(label=labels),
    link=dict(
        source=sources,
        target=targets,
        value=values,
        hovertemplate=(
            "%{source.label} → %{target.label}<br>"
            "Area: %{value:.2f} hectares"
            "<extra></extra>"
        )
    )
))

fig_subset.update_layout(
    title="Wetland Area in Hectares for California Wetlands larger than 1ha by Cowardin System, Subsystem, and Class",
    font_size=11,
    width=1100,
    height=700
)

fig_subset.show()
